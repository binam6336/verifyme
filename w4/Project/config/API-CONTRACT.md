                        📦 ساختار پاسخ (Response)
═══════════════════════════════════════════════════════════════════════════════

  ┌─────────────────────────────────────────────────────────────────────────┐
  │  وضعیت        │  ساختار پاسخ                                           │
  ├───────────────┼────────────────────────────────────────────────────────┤
  │  موفقیت آمیز  │  {                                                     │
  │  (success)    │    "status": "success",                                │
  │               │    "message": "پیام توضیحی",                           │
  │               │    "data": { ... }   // می‌تواند null باشد             │
  │               │  }                                                     │
  ├───────────────┼────────────────────────────────────────────────────────┤
  │  خطا          │  {                                                     │
  │  (error)      │    "status": "error",                                  │
  │               │    "message": "پیام خطای کلی",                         │
  │               │    "errors": {         // ✅ فقط در حالت خطا           │
  │               │      "field_name": "توضیح خطای این فیلد"              │
  │               │    }                                                   │
  │               │  }                                                     │
  └─────────────────────────────────────────────────────────────────────────┘

  ⚠️  فیلد "errors" فقط در حالت خطا (status: "error") ارسال می‌شود.
  ⚠️  فرانت‌اند موظف است اولین خطا را از شیء "errors" به کاربر نمایش دهد.

                        ۱. ثبت‌نام (Register)
═══════════════════════════════════════════════════════════════════════════════

  ┌─────────────────────────────────────────────────────────────────────────┐
  │  مشخصات  │  توضیح                                                     │
  ├──────────┼────────────────────────────────────────────────────────────┤
  │  آدرس    │  POST /auth/register/                                      │
  │  توکن    │  ❌ نیازی نیست (صفحه عمومی)                                │
  │  هدرها   │  Content-Type: application/json                            │
  │          │  Accept: application/json                                   │
  └─────────────────────────────────────────────────────────────────────────┘

  ── درخواست (Request Body) ──────────────────────────────────────────────
  {
    "first_name": "علی",          // الزامی - حداقل ۲ کاراکتر
    "last_name": "محمدی",          // الزامی - حداقل ۲ کاراکتر
    "company_name": "شرکت نمونه",  // الزامی - حداقل ۳ کاراکتر
    "email": "ali@example.com",    // الزامی - فرمت ایمیل معتبر
    "mobile": "09123456789",       // الزامی - ۱۱ رقم و شروع با 09
    "password": "12345678"         // الزامی - حداقل ۶ کاراکتر
  }

  ── پاسخ موفق (status: "success") ─────────────────────────────────────
  {
    "status": "success",
    "message": "ثبت‌نام با موفقیت انجام شد.",
    "data": {
      "token": "jwt_token_here",
      "user": {
        "id": 1,
        "first_name": "علی",
        "last_name": "محمدی",
        "company_name": "شرکت نمونه",
        "email": "ali@example.com",
        "mobile": "09123456789"
      }
    }
  }

  ── پاسخ خطا (status: "error") ────────────────────────────────────────
  {
    "status": "error",
    "message": "خطا در ثبت‌نام",
    "errors": {
      "email": "این ایمیل قبلاً ثبت شده است.",
      "mobile": "شماره موبایل نامعتبر است."
    }
  }

  ⚠️  فرانت‌اند بعد از دریافت توکن، آن را در localStorage ذخیره می‌کند.
  ⚠️  فرانت‌اند کاربر را به صفحه داشبورد هدایت می‌کند.

                    ۳. مقداردهی اولیه داشبورد (Dashboard Init)
═══════════════════════════════════════════════════════════════════════════════

  ┌─────────────────────────────────────────────────────────────────────────┐
  │  مشخصات  │  توضیح                                                     │
  ├──────────┼────────────────────────────────────────────────────────────┤
  │  آدرس    │  POST /dashboard/init/                                     │
  │  توکن    │  ✅ الزامی - در هدر Authorization                          │
  │  هدرها   │  Content-Type: application/json                            │
  │          │  Accept: application/json                                   │
  │          │  Authorization: <TOKEN>  ⚠️ فقط خود توکن                  │
  └─────────────────────────────────────────────────────────────────────────┘

  ── درخواست (Request Body) ──────────────────────────────────────────────
  {}  (بدنه خالی)

  ── پاسخ موفق (status: "success") ─────────────────────────────────────
  {
    "status": "success",
    "data": {
      "user": {
        "name": "علی محمدی",          // نام کامل کاربر
        "company_name": "شرکت نمونه",  // ✅ نام شرکت (برای نمایش زیر نام)
        "role": "تولیدکننده",          // نقش کاربر
        "avatar": "https://..."        // آدرس آواتار (اختیاری - می‌تواند null باشد)
      },
      "stats": {
        "total_products": 42,          // تعداد کل محصولات
        "active_warranties": 18,       // تعداد گارانتی‌های فعال
        "pending_activations": 7       // تعداد در انتظار فعال‌سازی
      }
    }
  }

  ── پاسخ خطا (status: "error") ────────────────────────────────────────
  {
    "status": "error",
    "message": "خطا در دریافت اطلاعات داشبورد",
    "errors": {
      "token": "توکن نامعتبر یا منقضی شده است."
    }
  }

  ⚠️  فیلد "company_name" برای نمایش نام شرکت در هدر استفاده می‌شود.
  ⚠️  در صورت خطای 401، فرانت‌اند توکن را پاک کرده و به صفحه ورود می‌رود.

                      📋 خلاصه رفتار فرانت‌اند با پاسخ‌ها
═══════════════════════════════════════════════════════════════════════════════

  ┌─────────────────────────────────────────────────────────────────────────┐
  │  وضعیت           │  اقدام فرانت‌اند                                   │
  ├───────────────────┼────────────────────────────────────────────────────┤
  │  status: success  │  ✅ نمایش پیام موفقیت (از فیلد "message")          │
  │                   │  ✅ استفاده از داده‌های "data"                     │
  │                   │  ✅ در صورت وجود توکن، ذخیره در localStorage        │
  ├───────────────────┼────────────────────────────────────────────────────┤
  │  status: error    │  ❌ نمایش خطا به کاربر                              │
  │                   │  ✅ اگر "errors" وجود داشت، اولین خطا را نمایش دهد  │
  │                   │  ✅ اگر "errors" وجود نداشت، از "message" استفاده  │
  │                   │  ✅ اگر خطای 401 بود، توکن را پاک و به ورود برود   │
  └─────────────────────────────────────────────────────────────────────────┘

                      ⚠️ کدهای وضعیت HTTP (اختیاری)
═══════════════════════════════════════════════════════════════════════════════

  ┌─────────────────────────────────────────────────────────────────────────┐
  │  کد HTTP   │  معنی                                                   │
  ├────────────┼──────────────────────────────────────────────────────────┤
  │  200       │  درخواست موفق (همراه با status: success در بدنه)       │
  │  400       │  خطای اعتبارسنجی (همراه با status: error در بدنه)      │
  │  401       │  توکن نامعتبر یا منقضی (فرانت‌اند باید logout کند)    │
  │  404       │  آدرس endpoint پیدا نشد                                 │
  │  500       │  خطای داخلی سرور                                        │
  └─────────────────────────────────────────────────────────────────────────┘

  ⚠️  فرانت‌اند همیشه محتوای JSON را بررسی می‌کند، حتی اگر کد HTTP خطا باشد.
  ⚠️  در صورت خطای 401، فرانت‌اند توکن را پاک کرده و به صفحه ورود می‌رود.
